		t.Case(fmt.Sprintf("cost = %v", c.cost), c.want, func() any {
			return minCostClimbingStairs(JClone(c.cost))
		})
	}
	// случайные против простого перебора всех наборов шагов
	brute := func(cost []int) int {
		n := len(cost)
		best := 1 << 60
		var rec func(pos, sum int)
		rec = func(pos, sum int) {
			if pos >= n {
				if sum < best {
					best = sum
				}
				return
			}
			rec(pos+1, sum+cost[pos])
			rec(pos+2, sum+cost[pos])
		}
		rec(0, 0)
		rec(1, 0)
		return best
	}
	seed := uint32(12345)
	rnd := func(m int) int {
		seed = seed*1664525 + 1013904223
		return int(seed>>8) % m
	}
	for k := 0; k < 5; k++ {
		n := 2 + rnd(14)
		cost := make([]int, n)
		for i := range cost {
			cost[i] = rnd(20)
		}
		cc := cost
		t.Case(fmt.Sprintf("cost = %v", cc), brute(cc), func() any {
			return minCostClimbingStairs(JClone(cc))

func judgeTests(t *Judge) {
	isPal := func(x string) bool {
		for i, j := 0, len(x)-1; i < j; i, j = i+1, j-1 {
			if x[i] != x[j] {
				return false
			}
		}
		return true
	}
	bestLen := func(s string) int {
		best := 0
		for i := 0; i < len(s); i++ {
			for j := i + 1; j <= len(s); j++ {
				if j-i > best && isPal(s[i:j]) {
					best = j - i
				}
			}
		}
		return best
	}
	run := func(s string) {
		want := bestLen(s)
		in := fmt.Sprintf("s = %q", s)
		if len(s) > 60 {
			in = fmt.Sprintf("s = строка длины %d, начинается с %q", len(s), s[:20])
		}
		t.Check(in, fmt.Sprintf("палиндром-подстрока s длины %d", want), func() (any, bool) {
			got := longestPalindrome(s)
			ok := len(got) == want && strings.Contains(s, got) && isPal(got)
			return got, ok
		})
	}
	for _, s := range []string{
		"babad", "cbbd", "a",
		"ab", "aa", "aaaa", "abcd", "forgeeksskeegfor", "abacdfgdcaba", "12321", "a1b2b1a", "bb", "abb",
	} {
		run(s)
	}
	seed := uint32(99)

	bfs := func(coins []int, amount int) int {
		dist := make([]int, amount+1)
		for i := range dist {
			dist[i] = -1
		}
		dist[0] = 0
		q := []int{0}
		for len(q) > 0 {
			x := q[0]
			q = q[1:]
			for _, c := range coins {
				y := x + c
				if y <= amount && dist[y] < 0 {
					dist[y] = dist[x] + 1
					q = append(q, y)
				}
			}
		}
		return dist[amount]
	}
	seed := uint32(555)
	rnd := func(m int) int {
		seed = seed*1664525 + 1013904223
		return int(seed>>8) % m
	}
	for k := 0; k < 6; k++ {
		n := 1 + rnd(5)
		coins := make([]int, n)
		for i := range coins {
			coins[i] = 2 + rnd(30)
		}
		amount := rnd(200)
		cs := coins
		t.Case(fmt.Sprintf("coins = %v, amount = %d", cs, amount), bfs(cs, amount), func() any {
