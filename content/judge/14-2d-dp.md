			return 0

		if amount >= coins[0] {

		}

			dp[i][0] = 1
		}
		for i := 1; i <= len(coins); i++ {
			for a := 1; a <= amount; a++ {
				dp[i][a] = dp[i-1][a]
				if a >= coins[i-1] {
					dp[i][a] += dp[i][a-coins[i-1]]
				}
			}
		}
		return dp[len(coins)][amount]
	}

func judgeTests(t *Judge) {
	cases := []struct {
		nums []int

	}{

		{[]int{2, 3}, 9},

		{[]int{7, 9, 8, 0, 7, 1, 3, 5, 5, 2, 3}, 1654},
	}

		t.Case(fmt.Sprintf("nums = %v", c.nums), c.want, func() any {
			return maxCoins(JClone(c.nums))
		})
	}
	// перебор порядка лопания (n <= 7)
	var brute func(a []int) int
	brute = func(a []int) int {
		if len(a) == 0 {
			return 0
		}
		best := 0
		for i := range a {
			l, r := 1, 1
			if i > 0 {
				l = a[i-1]
			}
			if i+1 < len(a) {
				r = a[i+1]
			}
			rest := make([]int, 0, len(a)-1)
			rest = append(rest, a[:i]...)
			rest = append(rest, a[i+1:]...)
			if v := l*a[i]*r + brute(rest); v > best {
				best = v
			}
		}
		return best
	}
	// интервальное DP "последний лопнутый шар" для больших n
	table := func(a []int) int {
		n := len(a)
		v := make([]int, n+2)
		v[0], v[n+1] = 1, 1
		copy(v[1:], a)
		dp := make([][]int, n+2)
		for i := range dp {
			dp[i] = make([]int, n+2)
		}
		for length := 2; length <= n+1; length++ {
			for l := 0; l+length <= n+1; l++ {
				r := l + length
				for k := l + 1; k < r; k++ {
					if x := dp[l][k] + dp[k][r] + v[l]*v[k]*v[r]; x > dp[l][r] {
						dp[l][r] = x
					}
				}
			}
		}
		return dp[0][n+1]
	}
	seed := uint32(1111)
	rnd := func(k int) int {
		seed = seed*1664525 + 1013904223
		return int(seed>>8) % k
	}

		n := 1 + rnd(7)
		a := make([]int, n)
		for i := range a {
			a[i] = rnd(101)
		}
		aa := a
		t.Case(fmt.Sprintf("nums = %v", aa), brute(aa), func() any {
			return maxCoins(JClone(aa))
		})
	}
